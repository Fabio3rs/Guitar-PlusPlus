if(BUILD_AUDIO_STUDIO OR BUILD_SFXGEN)
    # Reuse bundled Lua without CLuaH, gameplay registrations or graphics.
    # The state opens no libraries; lauxlib supplies the protected text loader.
    set(SFX_LUA_CORE lapi lauxlib lcode lctype ldebug ldo ldump lfunc lgc llex
        lmem lobject lopcodes lparser lstate lstring ltable ltm lundump lvm lzio)
    set(SFX_LUA_SOURCES)
    foreach(unit IN LISTS SFX_LUA_CORE)
        list(APPEND SFX_LUA_SOURCES "${PROJECT_SOURCE_DIR}/src/Lua/${unit}.c")
    endforeach()
    add_library(gpp_sfx_lua STATIC ${SFX_LUA_SOURCES})
    add_library(gpp_sfx STATIC sfx.cpp sfx_lua.cpp)
    target_link_libraries(gpp_sfx PUBLIC gpp_audio PRIVATE gpp_sfx_lua)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(gpp_sfx PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    if(BUILD_TESTING)
        find_package(GTest REQUIRED)
        add_executable(gpp_sfx_tests ${PROJECT_SOURCE_DIR}/tests/sfx_tests.cpp)
        target_link_libraries(gpp_sfx_tests PRIVATE gpp_sfx GTest::GTest GTest::Main)
        target_compile_definitions(gpp_sfx_tests PRIVATE GPP_SFX_SPECS="${PROJECT_SOURCE_DIR}/data/sounds/specs")
        add_test(NAME sfx_core COMMAND gpp_sfx_tests)
        set_tests_properties(sfx_core PROPERTIES LABELS "audio;sfx;cpu" TIMEOUT 120)
    endif()
endif()
